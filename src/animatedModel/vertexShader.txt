#version 400 core

const int MAX_JOINTS = 50; // Maximum joints allowed in a skeleton
const int MAX_WEIGHTS = 3; // Maximum number of joints that can affect a vertex

in vec3 position;
in vec2 textureCoords;
in vec3 normal;
in ivec3 jointIndices;
in vec3 weights;

out vec2 pass_textureCoords;
out vec3 surfaceNormal;
out vec3 toLightVector[20];
out vec3 toCameraVector;
out float visibility;

uniform mat4 jointTransforms[MAX_JOINTS];
uniform mat4 transformationMatrix;
uniform mat4 projectionMatrix;
uniform mat4 viewMatrix;
uniform vec3 lightPosition[20];
uniform vec3 cameraPosition;

uniform float useFakeLighting;
uniform float numberOfRows;
uniform vec2 offset;
uniform vec4 clipPlane;

const float density = 0.0035;
const float gradient = 5.0;

void main(void) {
    // Blending joint transforms
    vec4 totalLocalPos = vec4(0.0);
    vec4 totalNormal = vec4(0.0);

    for (int i = 0; i < MAX_WEIGHTS; i++) {
        mat4 jointTransform = jointTransforms[jointIndices[i]];
        totalLocalPos += (jointTransform * vec4(position, 1.0)) * weights[i];
        totalNormal += (jointTransform * vec4(normal, 0.0)) * weights[i];
    }

    // World-space transformations
    vec4 worldPosition = transformationMatrix * totalLocalPos;

    // Pass transformed position to the pipeline
    gl_Position = projectionMatrix * viewMatrix * worldPosition;

    // Texture coordinate manipulation for animation frames
    pass_textureCoords = textureCoords;

    // Clipping plane for reflections
    gl_ClipDistance[0] = dot(worldPosition, clipPlane);

    // Adjust normals
    vec3 actualNormal = totalNormal.xyz;
    if (useFakeLighting > 0.5) {
        actualNormal = vec3(0.0, 1.0, 0.0); // Use vertical normal for fake lighting
    }
    surfaceNormal = (transformationMatrix * vec4(normalize(actualNormal), 0.0)).xyz;

    // Calculate vectors to lights
    for (int i = 0; i < 20; i++) {
        toLightVector[i] = lightPosition[i] - worldPosition.xyz;
    }

    // Calculate vector to the camera
    toCameraVector = cameraPosition - worldPosition.xyz;

    // Fog effect visibility calculation
    float distance = length((viewMatrix * worldPosition).xyz);
    visibility = exp(-pow((distance * density), gradient));
    visibility = clamp(visibility, 0.0, 1.0);
}
